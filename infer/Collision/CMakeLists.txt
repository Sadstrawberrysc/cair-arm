cmake_minimum_required(VERSION 3.16)
project(uspilot_collision LANGUAGES CXX)

find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(EXPAT REQUIRED)
find_package(OpenSSL REQUIRED)
option(BUILD_COLLISION_CHECKING "Build offline FCL solid collision and segment checks" OFF)
option(BUILD_COLLISION_MONITOR "Build read-only RGB-D monitor and replay" OFF)
option(BUILD_COLLISION_PLANNING "Build offline OMPL joint planning" OFF)
option(BUILD_REALSENSE_SOURCE "Enable live D455 source (requires C++ librealsense)" OFF)
option(BUILD_MONITOR_VIEWER "Enable shared GLFW/OpenGL monitor and planning viewer" ON)
if((BUILD_COLLISION_MONITOR OR BUILD_REALSENSE_SOURCE OR BUILD_COLLISION_PLANNING) AND NOT BUILD_COLLISION_CHECKING)
    message(FATAL_ERROR "Monitor and planning require BUILD_COLLISION_CHECKING=ON")
endif()
if(BUILD_REALSENSE_SOURCE AND NOT BUILD_COLLISION_MONITOR)
    message(FATAL_ERROR "Live source requires BUILD_COLLISION_MONITOR=ON")
endif()

add_library(collision_geometry
    src/robot_model.cpp
    src/mesh.cpp
)
target_compile_features(collision_geometry PUBLIC cxx_std_17)
target_include_directories(collision_geometry PUBLIC include)
target_link_libraries(collision_geometry PUBLIC Eigen3::Eigen PRIVATE EXPAT::EXPAT OpenSSL::Crypto)
target_compile_options(collision_geometry PRIVATE -Wall -Wextra -Wpedantic)

# Offline comparison only: reuse the original MDH implementation without
# linking transport, Redis, force acquisition or any motion entrypoint.
add_library(collision_mdh_reference STATIC ../Robot/src/realman_kinematics.cpp)
target_include_directories(collision_mdh_reference PUBLIC ../Robot/include)
target_link_libraries(collision_mdh_reference PUBLIC Eigen3::Eigen)
target_compile_features(collision_mdh_reference PUBLIC cxx_std_17)

add_executable(collision_inspect apps/collision_inspect.cpp)
target_link_libraries(collision_inspect PRIVATE collision_geometry collision_mdh_reference)
target_compile_options(collision_inspect PRIVATE -Wall -Wextra -Wpedantic)

if(BUILD_COLLISION_CHECKING)
    find_package(fcl 0.7 REQUIRED CONFIG)
    find_package(Qhull REQUIRED CONFIG)
    add_library(collision_checking src/collision_checker.cpp src/voxel_scene.cpp)
    target_compile_features(collision_checking PUBLIC cxx_std_17)
    target_include_directories(collision_checking PUBLIC include)
    target_link_libraries(collision_checking PUBLIC collision_geometry PRIVATE fcl Qhull::qhull_r OpenSSL::Crypto)
    target_compile_options(collision_checking PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(collision_check apps/collision_check.cpp)
    target_include_directories(collision_check PRIVATE ../Robot/include)
    target_link_libraries(collision_check PRIVATE collision_checking fcl)
    target_compile_options(collision_check PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_COLLISION_PLANNING)
    find_package(ompl 1.5 REQUIRED CONFIG)
    find_path(COLLISION_BOOST_INCLUDE_DIR boost/concept_check.hpp HINTS "${OMPL_INCLUDE_DIR}/..")
    if(NOT COLLISION_BOOST_INCLUDE_DIR)
        message(FATAL_ERROR "OMPL requires Boost development headers")
    endif()
    # Some distribution configs embed absolute Boost paths. Use OMPL's shared
    # library directly so a user-extracted, relocatable prefix also works.
    find_library(COLLISION_OMPL_LIBRARY NAMES ompl HINTS ${OMPL_LIBRARY_DIRS})
    if(NOT COLLISION_OMPL_LIBRARY)
        message(FATAL_ERROR "OMPL shared library is missing")
    endif()
    add_library(collision_planning src/joint_planner.cpp src/pose_ik.cpp)
    target_compile_features(collision_planning PUBLIC cxx_std_17)
    target_include_directories(collision_planning PUBLIC include)
    target_include_directories(collision_planning SYSTEM PRIVATE ${COLLISION_BOOST_INCLUDE_DIR})
    foreach(ompl_include IN LISTS OMPL_INCLUDE_DIRS)
        if(IS_DIRECTORY "${ompl_include}")
            target_include_directories(collision_planning SYSTEM PRIVATE "${ompl_include}")
        endif()
    endforeach()
    target_link_libraries(collision_planning PUBLIC collision_checking PRIVATE ${COLLISION_OMPL_LIBRARY})
    target_compile_options(collision_planning PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(collision_plan apps/collision_plan.cpp)
    target_include_directories(collision_plan PRIVATE ../Robot/include ${OMPL_INCLUDE_DIR})
    target_include_directories(collision_plan SYSTEM PRIVATE ${COLLISION_BOOST_INCLUDE_DIR})
    target_link_libraries(collision_plan PRIVATE collision_planning ${COLLISION_OMPL_LIBRARY})
    target_compile_options(collision_plan PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_MONITOR_VIEWER AND (BUILD_COLLISION_MONITOR OR BUILD_COLLISION_PLANNING))
    find_package(OpenGL REQUIRED)
    find_package(glfw3 3.3 REQUIRED)
    add_library(collision_viewer apps/monitor_viewer.cpp)
    target_link_libraries(collision_viewer PUBLIC collision_checking PRIVATE fcl OpenGL::GL glfw)
    target_include_directories(collision_viewer PUBLIC apps)
    if(BUILD_COLLISION_PLANNING)
        target_link_libraries(collision_plan PRIVATE collision_viewer)
        target_compile_definitions(collision_plan PRIVATE COLLISION_HAS_VIEWER=1)
    endif()
endif()

if(BUILD_COLLISION_MONITOR)
    find_package(Threads REQUIRED)
    add_library(collision_monitor_transport STATIC ../Robot/src/realman_command.cpp)
    target_include_directories(collision_monitor_transport PUBLIC ../Robot/include)
    target_link_libraries(collision_monitor_transport PUBLIC Eigen3::Eigen Threads::Threads)
    target_compile_features(collision_monitor_transport PUBLIC cxx_std_17)
    add_library(collision_monitoring src/monitor.cpp src/robot_read_only.cpp)
    target_include_directories(collision_monitoring PUBLIC include PRIVATE ../Robot/include)
    target_link_libraries(collision_monitoring PUBLIC collision_checking PRIVATE collision_monitor_transport fcl)
    target_compile_options(collision_monitoring PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_REALSENSE_SOURCE)
        set(COLLISION_REALSENSE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/librealsense/include" CACHE PATH
            "Downloaded SDK include directory containing librealsense2/rs.hpp")
        set(COLLISION_REALSENSE_LIBRARY "" CACHE FILEPATH "Prebuilt librealsense2 shared library (SDK is never built)")
        if(COLLISION_REALSENSE_LIBRARY)
            if(NOT EXISTS "${COLLISION_REALSENSE_INCLUDE_DIR}/librealsense2/rs.hpp"
                OR NOT EXISTS "${COLLISION_REALSENSE_LIBRARY}")
                message(FATAL_ERROR "Provide an existing SDK include directory and prebuilt librealsense2 library")
            endif()
            add_library(collision_realsense SHARED IMPORTED)
            set_target_properties(collision_realsense PROPERTIES
                IMPORTED_LOCATION "${COLLISION_REALSENSE_LIBRARY}"
                INTERFACE_INCLUDE_DIRECTORIES "${COLLISION_REALSENSE_INCLUDE_DIR}")
            set(collision_realsense_target collision_realsense)
            message(STATUS "Using downloaded RealSense headers: ${COLLISION_REALSENSE_INCLUDE_DIR}")
            message(STATUS "Linking prebuilt RealSense library: ${COLLISION_REALSENSE_LIBRARY}")
        else()
            find_package(realsense2 QUIET CONFIG)
            if(NOT TARGET realsense2::realsense2)
                message(FATAL_ERROR "RealSense shared library missing. The downloaded source tree alone is not a runtime. Set COLLISION_REALSENSE_LIBRARY to a prebuilt librealsense2.so; SDK compilation is disabled.")
            endif()
            set(collision_realsense_target realsense2::realsense2)
        endif()
        target_sources(collision_monitoring PRIVATE src/live_source.cpp)
        target_link_libraries(collision_monitoring PRIVATE ${collision_realsense_target})
    endif()
    add_executable(collision_monitor apps/collision_monitor.cpp)
    target_include_directories(collision_monitor PRIVATE ../Robot/include)
    target_link_libraries(collision_monitor PRIVATE collision_monitoring fcl)
    target_compile_options(collision_monitor PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_COLLISION_PLANNING)
        target_link_libraries(collision_monitor PRIVATE collision_planning)
        target_include_directories(collision_monitor SYSTEM PRIVATE ${OMPL_INCLUDE_DIR} ${COLLISION_BOOST_INCLUDE_DIR})
        target_compile_definitions(collision_monitor PRIVATE COLLISION_HAS_PLANNING=1)
    endif()
    if(BUILD_REALSENSE_SOURCE)
        target_compile_definitions(collision_monitor PRIVATE COLLISION_HAS_REALSENSE=1)
        target_link_libraries(collision_monitor PRIVATE ${collision_realsense_target})
    endif()
    if(BUILD_MONITOR_VIEWER)
        target_link_libraries(collision_monitor PRIVATE collision_viewer)
        target_compile_definitions(collision_monitor PRIVATE COLLISION_HAS_VIEWER=1)
    endif()
endif()
